date | value

# ========================================
# 1. VALID VALUES
# 정상적인 value 확인
# ========================================

2011-01-03 | 3
2011-01-03 | 0
2011-01-03 | 1.5
2011-01-03 | 999
2011-01-03 | 1000


# ========================================
# 2. INVALID VALUES
# value는 0 이상 1000 이하여야 함
# ========================================

# negative value
2011-01-03 | -1

# value > 1000
2011-01-03 | 1001

# very large number
2011-01-03 | 2147483648


# ========================================
# 3. DATE LOOKUP
# DB에서 날짜를 올바르게 찾는지 확인
# ========================================

# exact date
2011-01-03 | 1

# date가 DB에 없을 경우
# → 가장 가까운 이전(lower) 날짜를 사용해야 함
2011-01-04 | 2

# 다른 날짜 검색 테스트
2011-01-09 | 1


# ========================================
# 4. DATE BOUNDARIES
# DB 범위의 시작/끝 주변 확인
# 실제 data.csv를 보고 나중에 날짜 조정하기
# ========================================

# DB의 최초 날짜보다 이전일 가능성이 있는 날짜
2009-01-02 | 1

# 비교적 최근 날짜
2022-03-29 | 1


# ========================================
# 5. LEAP YEAR
# 윤년 / 2월 29일 검증
# ========================================

# valid leap year date
2012-02-29 | 2

# invalid: 2011 is not a leap year
2011-02-29 | 2


# ========================================
# 6. INVALID CALENDAR DATES
# 형식은 비슷하지만 실제 존재하지 않는 날짜
# ========================================

# invalid month
2011-13-01 | 2
2011-00-01 | 2

# April has only 30 days
2011-04-31 | 2

# invalid day
2011-01-32 | 2
2011-01-00 | 2

# completely invalid date
2001-42-42 | 1


# ========================================
# 7. INVALID DATE FORMAT
# YYYY-MM-DD 형식이 아닌 경우
# ========================================

2001-42-42
2011/01/03 | 2
11-01-03 | 2
20110103 | 2
2011-1-03 | 2
2011-01-3 | 2


# ========================================
# 8. MISSING DATA
# date / separator / value가 빠진 경우
# ========================================

# missing separator and value
2011-01-03

# missing value
2011-01-03 |

# missing date
| 3


# ========================================
# 9. INVALID NUMBER FORMAT
# value를 정상적인 숫자로 해석할 수 있는지 확인
# ========================================

# text instead of number
2011-01-03 | abc

# number followed by text
2011-01-03 | 3abc

# malformed floating point number
2011-01-03 | 3.14.15


# ========================================
# 10. SPECIAL NUMBER FORMATS
# 우리가 어떤 형식까지 valid로 인정할지 확인
# ========================================

# explicit positive sign
2011-01-03 | +3

# decimal between 0 and 1
2011-01-03 | 0.5

# leading zeros
2011-01-03 | 0003

# 11. COMPLETELY INVALID INPUT
